// AI INTEGRATION MODULE // Kết nối AI module với hệ thống chính class AIIntegration { constructor() { this.aiClient = null; this.isIntegrated = false; } // Tích hợp với hệ thống integrateWithSystem() { console.log("🔄 Đang tích hợp AI với hệ thống..."); // Kiểm tra cấu hình AI if (!window.AI_CONFIG) { console.error("❌ Không tìm thấy cấu hình AI"); return false; } // Khởi tạo AI Client this.aiClient = new window.AIClient(window.AI_CONFIG); this.aiClient.initialize(); // Tích hợp sự kiện this.setupEventListeners(); // Kiểm tra cảnh báo dùng thử this.checkTrialReminder(); this.isIntegrated = true; console.log("✅ Đã tích hợp AI với hệ thống"); return true; } // Thiết lập event listeners setupEventListeners() { // Lắng nghe sự kiện từ hệ thống document.addEventListener('ai_send_message', (event) => { this.handleAIMessage(event.detail.message); }); // Lắng nghe sự kiện hết hạn dùng thử document.addEventListener('trial_warning', (event) => { this.handleTrialWarning(event.detail.daysLeft); }); // Lắng nghe sự kiện thanh toán thành công document.addEventListener('payment_success', (event) => { this.handlePaymentSuccess(event.detail); }); } // Xử lý tin nhắn AI async handleAIMessage(message) { if (!this.aiClient) { console.error("AI Client chưa được khởi tạo"); return; } try { const response = await this.aiClient.sendMessage(message); // Phát sự kiện phản hồi const responseEvent = new CustomEvent('ai_response', { detail: { response: response } }); document.dispatchEvent(responseEvent); } catch (error) { console.error("Lỗi xử lý tin nhắn AI:", error); } } // Xử lý cảnh báo hết hạn dùng thử handleTrialWarning(daysLeft) { if (!this.aiClient || !window.AI_CONFIG.REMINDER_ENABLED) return; const warningMessage = this.aiClient.sendTrialWarning(daysLeft); if (warningMessage) { // Hiển thị cảnh báo this.showAIToast(warningMessage); // Gửi webhook cảnh báo this.sendWarningToServer(daysLeft); } } // Xử lý thanh toán thành công handlePaymentSuccess(paymentData) { if (!this.aiClient) return; // Gửi tin nhắn chúc mừng const congratsMessage = `🎉 Chúc mừng! Bạn đã kích hoạt thành công gói ${paymentData.package}. Cảm ơn đã sử dụng dịch vụ!`; // Thêm vào lịch sử AI this.aiClient.addToHistory('ai', congratsMessage); // Hiển thị thông báo this.showAIToast(congratsMessage); } // Kiểm tra cảnh báo dùng thử checkTrialReminder() { // Kiểm tra dữ liệu dùng thử const trialData = localStorage.getItem('free_trial_data'); if (!trialData) return; const trial = JSON.parse(trialData); const endDate = new Date(trial.endDate); const now = new Date(); const diffTime = endDate - now; const daysLeft = Math.ceil(diffTime / (1000 * 60 * 60 * 24)); if (daysLeft <= window.AI_CONFIG.REMINDER_DAYS_BEFORE) { this.handleTrialWarning(daysLeft); } } // Hiển thị toast AI showAIToast(message) { // Tạo toast element nếu chưa có let aiToast = document.getElementById('ai-toast'); if (!aiToast) { aiToast = document.createElement('div'); aiToast.id = 'ai-toast'; aiToast.style.cssText = ` position: fixed; bottom: 100px; left: 50%; transform: translateX(-50%); background: linear-gradient(135deg, #00bfa5, #00897b); color: white; padding: 15px 25px; border-radius: 25px; z-index: 999999; display: none; font-weight: 600; box-shadow: 0 10px 25px rgba(0,0,0,0.2); max-width: 90%; text-align: center; animation: slideUp 0.3s ease-out; `; // Thêm CSS animation const style = document.createElement('style'); style.textContent = ` @keyframes slideUp { from { transform: translate(-50%, 100%); opacity: 0; } to { transform: translate(-50%, 0); opacity: 1; } } `; document.head.appendChild(style); document.body.appendChild(aiToast); } // Hiển thị toast aiToast.textContent = message; aiToast.style.display = 'block'; // Tự động ẩn sau 5 giây setTimeout(() => { aiToast.style.display = 'none'; }, 5000); } // Gửi cảnh báo đến server sendWarningToServer(daysLeft) { const warningData = { type: 'ai_trial_warning', days_left: daysLeft, user_id: localStorage.getItem('customer_id') || localStorage.getItem('taxi_user_id'), timestamp: new Date().toISOString() }; // Gửi đến webhook fetch(window.AI_CONFIG.AI_WEBHOOK, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(warningData) }).catch(error => { console.error("Lỗi gửi cảnh báo:", error); }); } // Gửi tin nhắn qua AI sendMessage(message) { if (!this.isIntegrated) { this.integrateWithSystem(); } return this.handleAIMessage(message); } // Lấy trạng thái tích hợp getIntegrationStatus() { return { integrated: this.isIntegrated, aiConnected: this.aiClient ? this.aiClient.checkConnection() : false, messageCount: this.aiClient ? this.aiClient.getMessageHistory().length : 0 }; } } // Tạo instance toàn cục if (typeof window !== 'undefined') { window.AIIntegration = AIIntegration; // Tự động tích hợp khi tải trang document.addEventListener('DOMContentLoaded', function() { // Chờ một chút để đảm bảo AI_CONFIG đã tải setTimeout(() => { if (window.AI_CONFIG) { window.aiIntegration = new AIIntegration(); window.aiIntegration.integrateWithSystem(); } }, 1000); }); } // Export cho Node.js if (typeof module !== 'undefined' && module.exports) { module.exports = AIIntegration; }
